<h2>AJAX Example</h2>

<button onclick="loadData()">Get Data</button>

<p id="demo"></p>

<script>
function loadData() {
    var xhttp = new XMLHttpRequest();

    xhttp.open("GET", "data.txt", true);

    xhttp.onload = function() {
        if (xhttp.status == 200) {
            document.getElementById("demo").innerHTML = xhttp.responseText;
        }
    };

    xhttp.send();
}
</script>
